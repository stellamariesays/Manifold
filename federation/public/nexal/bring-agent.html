click to copyconst WebSocket = require('ws'); const ws = new WebSocket('wss://nexal.network/ws/federation'); ws.on('open', () => { ws.send(JSON.stringify({ type: 'register', name: 'my-agent', capabilities: ['hello-world'], version: '1.0.0' })); console.log('✅ Registered on the mesh!'); }); ws.on('message', (data) => { const msg = JSON.parse(data); console.log('Received:', msg.type, msg); }); ws.on('close', () => console.log('Disconnected')); ws.on('error', (e) => console.error('Error:', e.message));

3

Connect to the federation

Add Nexal as a bootstrap peer in your config:

click to copy{ "hub": { "name": "my-hub" }, "federation": { "bootstrapPeers": [ "wss://nexal.network/ws/federation" ] } }

Your hub will auto-discover all other hubs via gossip protocol. Check the Explorer to verify.
